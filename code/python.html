"""dSPACE/Unityの改行区切り2値通信。実機APIと通信形式を分離する。"""
from __future__ import annotations

import json
import socket
import time
from pathlib import Path
from typing import Protocol


class Device(Protocol):
    def read_pair(self) -> tuple[float, float]: ...
    def write_pair(self, pair: tuple[float, float]) -> None: ...
    def close(self) -> None: ...


def encode_measurement(pair: tuple[float, float]) -> bytes:
    """元コードと同じ小数4桁・カンマ・LFで送信する。"""
    return f"{pair[0]:.4f},{pair[1]:.4f}\n".encode("utf-8")


class CommandBuffer:
    """部分行を保持し、1回の受信で届いた最後の候補行を処理する。"""
    def __init__(self):
        self.pending = ""

    def feed(self, chunk: bytes) -> tuple[float, float] | None:
        self.pending += chunk.decode("utf-8")
        lines = self.pending.split("\n")
        if self.pending.endswith("\n"):
            self.pending = ""
        else:
            self.pending = lines[-1]
            lines = lines[:-1]
        # 元実装通り、最後のカンマ付き行が不正なら古い行へ戻らない。
        candidate = next((line.strip() for line in reversed(lines) if "," in line.strip()), None)
        if candidate is None:
            return None
        parts = candidate.split(",")
        if len(parts) != 2:
            return None
        try:
            return float(parts[0]), float(parts[1])
        except ValueError:
            return None


def exchange_once(sock: socket.socket, device: Device, parser: CommandBuffer):
    values = device.read_pair()
    sock.sendall(encode_measurement(values))
    command = None
    try:
        data = sock.recv(4096)
        if not data:
            raise ConnectionError("Peer closed the connection")
        command = parser.feed(data)
        if command is not None:
            device.write_pair(command)
    except socket.timeout:
        pass
    return values, command


class DSpaceDevice:
    """実機専用。pythonnetとdSPACE XIL 2.2の導入は別途必要。"""
    def __init__(self, config: dict):
        import clr
        clr.AddReference("ASAM.XIL.Implementation.TestbenchFactory, Version=2.2.0.0, Culture=neutral, PublicKeyToken=bf471dff114ae984")
        clr.AddReference("ASAM.XIL.Interfaces, Version=2.2.0.0, Culture=neutral, PublicKeyToken=bf471dff114ae984")
        from ASAM.XIL.Implementation.TestbenchFactory.Testbench import TestbenchFactory
        from ASAM.XIL.Interfaces.Testbench.Common.VariableRef.Enum import ValueRepresentation
        bench = TestbenchFactory().CreateVendorSpecificTestbench("dSPACE GmbH", "XIL API", config["release_version"])
        self.value_factory = bench.ValueFactory
        self.port = bench.MAPortFactory.CreateMAPort("DemoMAPort")
        self.port.Configure(self.port.LoadConfiguration(config["maport_config_file"]), False)
        make_ref = lambda p: bench.VariableRefFactory.CreateGenericVariableRef(p, ValueRepresentation.ePhysicalValue)
        self.read_refs = [make_ref(p) for p in config["read_paths"]]
        self.write_refs = [make_ref(p) for p in config["write_paths"]]
        if len(self.read_refs) != 2 or len(self.write_refs) != 2:
            raise ValueError("Two read paths and two write paths are required")

    def read_pair(self):
        return tuple(float(self.port.Read2(ref).Value) for ref in self.read_refs)

    def write_pair(self, pair):
        for ref, value in zip(self.write_refs, pair):
            self.port.Write2(ref, self.value_factory.CreateFloatValue(value))

    def close(self):
        self.port.Dispose()


def run_real(config_file: Path, duration: float):
    """実機接続は明示的なrealサブコマンドからのみ実行。未実機検証。"""
    config = json.loads(config_file.read_text(encoding="utf-8"))
    device = DSpaceDevice(config)
    try:
        with socket.create_connection((config["host"], int(config["port"])), timeout=3.0) as sock:
            sock.settimeout(0.01)
            parser = CommandBuffer()
            stop = time.monotonic() + duration
            while time.monotonic() < stop:
                exchange_once(sock, device, parser)
                time.sleep(0.001)
    finally:
        device.close()

"""押し戻し区間の検出。入出力・図描画から計算を分離。"""
from __future__ import annotations
from typing import List, Tuple, Optional, Dict
import numpy as np
import pandas as pd
from .config import (EXPECTED_PRESS_COUNT, START_THRESHOLD_MM, BACKOFF_S, MIN_PRESS_EXCURSION_MM, MIN_PRESS_DURATION_S, MERGE_GAP_S, SMOOTH_WINDOW_S, BASELINE_PERCENTILE, FALLBACK_TO_GLOBAL_BASELINE, SUSPECT_MIN_PEAK_MM, SUSPECT_MIN_INTERPEAK_DROP_MM, SUSPECT_MAX_VALLEY_MM)
from .signal_utils import (safe_median_dt, contiguous_regions, merge_close_regions, smooth_signal, find_local_peaks)


def find_first_condi_drop_index(condi: np.ndarray) -> Optional[int]:
    """
    Condi(n) > Condi(n+1) となる最初の n を見つけ、
    その次のサンプル n+1 を「減少後開始位置」として返す。
    """
    c = np.asarray(condi, dtype=float)
    if len(c) < 2:
        return None
    diff = c[:-1] - c[1:]
    idx = np.where(np.isfinite(diff) & (diff > 0))[0]
    if len(idx) == 0:
        return None
    return int(idx[0] + 1)

def compute_lower5_mean_baseline(signal: np.ndarray, condi: np.ndarray, percentile: float=BASELINE_PERCENTILE, fallback_to_global: bool=FALLBACK_TO_GLOBAL_BASELINE) -> Tuple[float, str, Optional[int], int]:
    """
    Condi減少後区間の signal の下位5%値群の平均を baseline とする。
    戻り値:
      baseline_value, baseline_method, condi_drop_start_idx, baseline_n
    """
    x = np.asarray(signal, dtype=float)
    valid_all = np.isfinite(x)
    if np.sum(valid_all) == 0:
        return (np.nan, 'no_valid_signal', None, 0)
    drop_start_idx = find_first_condi_drop_index(condi)

    def lower_percentile_mean(vals: np.ndarray) -> Tuple[float, int]:
        vals = vals[np.isfinite(vals)]
        if len(vals) == 0:
            return (np.nan, 0)
        th = np.nanpercentile(vals, percentile)
        picked = vals[vals <= th]
        if len(picked) == 0:
            return (np.nan, 0)
        return (float(np.nanmean(picked)), int(len(picked)))
    if drop_start_idx is not None and drop_start_idx < len(x):
        tail = x[drop_start_idx:]
        base, n = lower_percentile_mean(tail)
        if n > 0:
            return (base, 'condi_drop_tail_lower5_mean', drop_start_idx, n)
    if fallback_to_global:
        base, n = lower_percentile_mean(x[valid_all])
        if n > 0:
            return (base, 'global_lower5_mean_fallback', drop_start_idx, n)
    return (np.nan, 'baseline_not_found', drop_start_idx, 0)

def detect_two_peaks_connected_in_region(disp_rel_region: np.ndarray, min_peak_mm: float=SUSPECT_MIN_PEAK_MM, min_interpeak_drop_mm: float=SUSPECT_MIN_INTERPEAK_DROP_MM, max_valley_mm: float=SUSPECT_MAX_VALLEY_MM) -> Dict[str, object]:
    x = np.asarray(disp_rel_region, dtype=float)
    result = {'suspect_connected': False, 'num_peaks_over_threshold': 0, 'peak1_idx_local': np.nan, 'peak2_idx_local': np.nan, 'peak1_mm': np.nan, 'peak2_mm': np.nan, 'valley_between_peaks_mm': np.nan, 'interpeak_drop_from_lower_peak_mm': np.nan}
    if len(x) < 3 or np.sum(np.isfinite(x)) < 3:
        return result
    peak_idx_all = find_local_peaks(x)
    peak_idx = [i for i in peak_idx_all if np.isfinite(x[i]) and x[i] >= min_peak_mm]
    result['num_peaks_over_threshold'] = len(peak_idx)
    if len(peak_idx) < 2:
        return result
    for p1, p2 in zip(peak_idx[:-1], peak_idx[1:]):
        if p2 <= p1 + 1:
            continue
        peak1 = float(x[p1])
        peak2 = float(x[p2])
        valley = float(np.nanmin(x[p1:p2 + 1]))
        lower_peak = min(peak1, peak2)
        interpeak_drop = lower_peak - valley
        if interpeak_drop >= min_interpeak_drop_mm and valley > max_valley_mm:
            result.update({'suspect_connected': True, 'peak1_idx_local': p1, 'peak2_idx_local': p2, 'peak1_mm': peak1, 'peak2_mm': peak2, 'valley_between_peaks_mm': valley, 'interpeak_drop_from_lower_peak_mm': interpeak_drop})
            return result
    return result

def detect_multi_press_in_region(disp_rel_region: np.ndarray, min_peak_mm: float=SUSPECT_MIN_PEAK_MM, min_interpeak_drop_mm: float=SUSPECT_MIN_INTERPEAK_DROP_MM) -> Dict[str, object]:
    x = np.asarray(disp_rel_region, dtype=float)

        return result

    peak_idx = [i for i in peak_idx_all if np.isfinite(x[i]) and x[i] >= min_peak_mm]

        return result

def detect_first_two_press_return_regions(df: pd.DataFrame, expected_count: int=EXPECTED_PRESS_COUNT, start_threshold_mm: float=START_THRESHOLD_MM, backoff_s: float=BACKOFF_S, min_excursion_mm: float=MIN_PRESS_EXCURSION_MM, min_duration_s: float=MIN_PRESS_DURATION_S, merge_gap_s: float=MERGE_GAP_S, smooth_window_s: float=SMOOTH_WINDOW_S) -> Tuple[List[Tuple[int, int]], Dict[str, object], List[Dict[str, object]]]:
    time_s = df['time[s]'].to_numpy(dtype=float)
    disp = df['x[mm]'].to_numpy(dtype=float)
    condi = df['Condi'].to_numpy(dtype=float)
    dt = safe_median_dt(time_s)
    if not np.isfinite(dt) or dt <= 0:
        baseline_info = {'baseline_mm': np.nan, 'baseline_method_mm': 'invalid_dt', 'condi_drop_start_idx': None, 'baseline_n_mm': 0}
        return ([], baseline_info, [])
    fs = 1.0 / dt
    win = max(3, int(round(fs * smooth_window_s)))
    disp_s = smooth_signal(disp, win)
    valid = np.isfinite(disp_s)
    if np.sum(valid) < 5:
        baseline_info = {'baseline_mm': np.nan, 'baseline_method_mm': 'not_enough_valid_disp', 'condi_drop_start_idx': None, 'baseline_n_mm': 0}
        return ([], baseline_info, [])
    baseline_mm, baseline_method_mm, condi_drop_start_idx, baseline_n_mm = compute_lower5_mean_baseline(signal=disp_s, condi=condi, percentile=BASELINE_PERCENTILE, fallback_to_global=FALLBACK_TO_GLOBAL_BASELINE)
    baseline_info = {'baseline_mm': baseline_mm, 'baseline_method_mm': baseline_method_mm, 'condi_drop_start_idx': condi_drop_start_idx, 'baseline_n_mm': baseline_n_mm}
    disp_rel = disp_s - baseline_mm
    active = disp_rel > start_threshold_mm
    regions = contiguous_regions(active)
    if not regions:
        return ([], baseline_info, [])
    gap_n = int(round(merge_gap_s / dt))
    merged = merge_close_regions(regions, gap_n)
    min_n = int(round(min_duration_s / dt))
    backoff_n = int(round(backoff_s / dt))
    valid_regions: List[Tuple[int, int, float]] = []
    suspect_info_list: List[Dict[str, object]] = []
    for s, e in merged:
        if e - s + 1 < min_n:
            continue
        seg = disp_rel[s:e + 1]
        if np.sum(np.isfinite(seg)) < 5:
            continue
        excursion = float(np.nanmax(seg) - np.nanmin(seg))
        if excursion < min_excursion_mm:
            continue
        s2 = max(0, s - backoff_n)
        valid_regions.append((s2, e, excursion))
        connected = detect_two_peaks_connected_in_region(seg)
        multi_press = detect_multi_press_in_region(seg)
        suspect_info_list.append({'RegionStartIndex_before_backoff': s, 'RegionStartIndex_after_backoff': s2, 'RegionEndIndex': e, 'RegionExcursion_mm': excursion, **connected, **multi_press, 'suspect_any': bool(connected.get('suspect_connected', False) or multi_press.get('suspect_multi_press', False))})
    if not valid_regions:
        return ([], baseline_info, suspect_info_list)
    valid_regions = sorted(valid_regions, key=lambda x: x[0])[:expected_count]
    return ([(s, e) for s, e, _ in valid_regions], baseline_info, suspect_info_list)

"""加速度包絡線と変位補間。元のHilbert計算を維持。"""
from __future__ import annotations
import numpy as np
import pandas as pd
from scipy.signal import hilbert


def compute_hilbert_envelope_mirror(x: np.ndarray, pad: int=100) -> np.ndarray:
    """
    ミラー拡張してから Hilbert を適用し、中央部のみ取り出す
    """
    x = np.asarray(x, dtype=float)
    valid = np.isfinite(x)
    if np.sum(valid) < 2:
        return np.full_like(x, np.nan, dtype=float)
    x_work = x.copy()
    if not np.all(valid):
        idx = np.arange(len(x_work))
        x_work[~valid] = np.interp(idx[~valid], idx[valid], x_work[valid])
    n = len(x_work)
    pad = int(min(max(pad, 1), max(n - 1, 1)))
    if n < 3:
        env = np.abs(hilbert(x_work))
        env[~valid] = np.nan
        return env
    left = x_work[1:pad + 1][::-1]
    right = x_work[-pad - 1:-1][::-1]
    x_pad = np.r_[left, x_work, right]
    env_pad = np.abs(hilbert(x_pad))
    env = env_pad[pad:pad + n]
    env[~valid] = np.nan
    return env

def interp_envelope_vs_displacement(disp: np.ndarray, env: np.ndarray, disp_common_raw: np.ndarray) -> np.ndarray:
    disp = np.asarray(disp, dtype=float)
    env = np.asarray(env, dtype=float)
    valid = np.isfinite(disp) & np.isfinite(env)
    if np.sum(valid) < 2:
        return np.full_like(disp_common_raw, np.nan, dtype=float)
    disp_valid = disp[valid]
    env_valid = env[valid]
    df_tmp = pd.DataFrame({'disp': disp_valid, 'env': env_valid})
    df_tmp = df_tmp.groupby('disp', as_index=False)['env'].mean().sort_values('disp')
    disp_unique = df_tmp['disp'].to_numpy(dtype=float)
    env_unique = df_tmp['env'].to_numpy(dtype=float)
    if len(disp_unique) < 2:
        return np.full_like(disp_common_raw, np.nan, dtype=float)
    y_interp = np.interp(disp_common_raw, disp_unique, env_unique)
    y_interp[disp_common_raw < disp_unique[0]] = np.nan
    y_interp[disp_common_raw > disp_unique[-1]] = np.nan
    return y_interp

"""CSVを丸め直さず、Python floatのround-trip精度で読み込む。"""
from pathlib import Path
import numpy as np
import pandas as pd


def _number(text: str) -> float:
    try:
        return float(text)
    except (TypeError, ValueError):
        return np.nan


def read_float_csv(path: Path, numeric_columns: list[str]) -> pd.DataFrame:
    raw = pd.read_csv(path, dtype=str, keep_default_na=False)
    missing = set(numeric_columns) - set(raw)
    if missing:
        raise ValueError(f"Missing columns: {sorted(missing)} in {path.name}")
    numeric = set(numeric_columns)
    return pd.DataFrame({name: raw[name].map(_number).to_numpy(dtype=float) if name in numeric else raw[name] for name in raw})

"""時系列の平滑化・連続区間処理。原実装の計算順を維持。"""
from __future__ import annotations
from typing import List, Tuple, Optional, Dict
import numpy as np


def safe_median_dt(time_s: np.ndarray) -> float:
    time_s = np.asarray(time_s, dtype=float)
    if len(time_s) < 2:
        return np.nan
    dt = np.diff(time_s)
    dt = dt[np.isfinite(dt) & (dt > 0)]
    if len(dt) == 0:
        return np.nan
    return float(np.median(dt))

def contiguous_regions(mask: np.ndarray) -> List[Tuple[int, int]]:
    if mask.size == 0:
        return []
    mask = mask.astype(bool)
    diff = np.diff(mask.astype(int))
    starts = list(np.where(diff == 1)[0] + 1)
    ends = list(np.where(diff == -1)[0])
    if mask[0]:
        starts = [0] + starts
    if mask[-1]:
        ends = ends + [len(mask) - 1]
    return list(zip(starts, ends))

def merge_close_regions(regions: List[Tuple[int, int]], gap_max_samples: int) -> List[Tuple[int, int]]:
    if not regions:
        return []
    merged = [regions[0]]
    for s, e in regions[1:]:
        ps, pe = merged[-1]
        if s - pe <= gap_max_samples:
            merged[-1] = (ps, e)
        else:
            merged.append((s, e))
    return merged

def smooth_signal(x: np.ndarray, window: int) -> np.ndarray:
    x = np.asarray(x, dtype=float)
    if x.size == 0 or window <= 1:
        return x.copy()
    pad = window // 2
    x_pad = np.pad(x, pad_width=pad, mode='edge')
    kernel = np.ones(window, dtype=float) / window
    y = np.convolve(x_pad, kernel, mode='valid')
    if len(y) != len(x):
        y = y[:len(x)]
    return y

def find_local_peaks(signal: np.ndarray) -> List[int]:
    x = np.asarray(signal, dtype=float)
    peaks = []
    if len(x) < 3:
        return peaks
    for i in range(1, len(x) - 1):
        a, b, c = (x[i - 1], x[i], x[i + 1])
        if not (np.isfinite(a) and np.isfinite(b) and np.isfinite(c)):
            continue
        if b >= a and b >= c:
            peaks.append(i)
    return peaks

def run_segmentation(data: Path, output: Path):
    files = sorted((data / "segmentation").glob("*.csv"))
    if not files:
        raise FileNotFoundError("No segmentation CSV files")
    output.mkdir(parents=True, exist_ok=True)

def process_envelope(path: Path):
    frame = read_numeric(path, ["time[s]", "x[mm]", "acc_high(80-120Hz)", "Condi"])
    code = frame["Condi"].dropna().astype(int)
    code = code[code != 0]
    condition = CONDITIONS.get(int(code.mode().iloc[0])) if len(code) else None
    if condition is None:
        return None
    disp = frame["x[mm]"].to_numpy(dtype=float)
    # 原コードの補正係数。既存の列名はm/s^2を示すが値は元処理のまま。
    acc = frame["acc_high(80-120Hz)"].to_numpy(dtype=float) / 0.102
    valid = np.isfinite(disp) & np.isfinite(acc)
    disp, acc = disp[valid], acc[valid]
    time_s = frame["time[s]"].to_numpy(dtype=float)[valid]
    if len(disp) < 3 or float(np.nanmax(disp)) < 84.0:
        return None
    env = compute_hilbert_envelope_mirror(acc, pad=100)
    mask = (disp >= 5.0) & (disp <= 85.0) & np.isfinite(env)
    if np.sum(mask) < 3:
        return None
    curve = interp_envelope_vs_displacement(disp[mask], env[mask], GRID_RAW)
    return {"participant": path.parent.name, "condition": condition, "curve": curve, "time_s": time_s, "acc": acc, "envelope": env}

class MockDevice:
    def __init__(self):
        self.index = 0
        self.command = (0.0, 0.0)

    def read_pair(self):
        t = self.index * 0.01
        self.index += 1
        return 40 + 30 * math.sin(t * 3), 35 + 20 * math.cos(t * 3)

    def write_pair(self, pair):
        self.command = pair

    def close(self):
        pass


def run_tcp_demo(output: Path, samples: int = 120):
    """localhost上で本物のTCPを使うが、センサ・指令は合成値。"""
    output.mkdir(parents=True, exist_ok=True)
    server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server.bind(("127.0.0.1", 0))
    server.listen(1)
    errors = []

    def mock_unity():
        try:
            with server.accept()[0] as peer:
                pending = ""
                count = 0
                while count < samples:
                    chunk = peer.recv(4096)
                    if not chunk:
                        break
                    pending += chunk.decode("utf-8")
                    while "\n" in pending:
                        line, pending = pending.split("\n", 1)
                        a, b = map(float, line.split(","))
                        response = f"{a / 80:.6f},{b / 80:.6f}\n".encode()
                        midpoint = len(response) // 2
                        peer.sendall(response[:midpoint])
                        peer.sendall(response[midpoint:])
                        count += 1
        except Exception as exc:
            errors.append(str(exc))
        finally:
            server.close()

    thread = threading.Thread(target=mock_unity, daemon=True)
    thread.start()
    rows = []
    device = MockDevice()
    with socket.create_connection(server.getsockname(), timeout=2.0) as client:
        client.settimeout(0.1)
        parser = CommandBuffer()
        for i in range(samples):
            values, command = exchange_once(client, device, parser)
            # デモは各要求の応答を読み切る。実機ループの周期を模倣しない。
            while command is None:
                chunk = client.recv(4096)
                if not chunk:
                    raise ConnectionError("Mock peer closed before a complete response")
                command = parser.feed(chunk)
                if command is not None:
                    device.write_pair(command)
            rows.append({"sample": i, "synthetic_time_s": i * 0.01, "mock_position_1": values[0], "mock_position_2": values[1], "mock_command_1": device.command[0], "mock_command_2": device.command[1], "command_received_this_step": command is not None})
    thread.join(timeout=3.0)
    if thread.is_alive() or errors:
        raise RuntimeError(f"Loopback server failed: {errors}")
    df = pd.DataFrame(rows)
    df.to_csv(output / "mock_tcp_trace.csv", index=False)
    fig, axes = plt.subplots(2, 1, figsize=(9, 6), sharex=True, layout="constrained")
    axes[0].plot(df["sample"], df["mock_position_1"], label="Mock sensor 1")
    axes[0].plot(df["sample"], df["mock_position_2"], label="Mock sensor 2")
    axes[0].set(ylabel="Synthetic displacement", title="MOCK ONLY: local TCP exchange, no dSPACE hardware")
    axes[1].plot(df["sample"], df["mock_command_1"], label="Mock command 1")
    axes[1].plot(df["sample"], df["mock_command_2"], label="Mock command 2")
    axes[1].set(xlabel="Exchange step (not measured real-time rate)", ylabel="Synthetic command")
    for ax in axes:
        ax.legend()
        ax.grid(alpha=0.2)
    save_figure(fig, output, "mock_tcp_exchange")
    return {"kind": "synthetic_device_with_real_loopback_tcp", "exchanges": samples, "received_commands": int(df["command_received_this_step"].sum()), "hardware_verified": False}
