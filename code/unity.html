using UnityEngine;

namespace Portfolio.Research
{
    /// <summary>研磨面を基準とした表示点・入力点を取得する掲載用サンプル。</summary>
    public sealed class RelativePoseProvider : MonoBehaviour
    {
        [SerializeField] private Transform origin;
        [SerializeField] private Transform visualPoint;
        [SerializeField] private Transform rawPoint;
        [SerializeField] private ContactFlag contact;
        [SerializeField, Min(1)] private int sampleRateHz = 100;
        [SerializeField] private float positionScale = 1000f;
        [SerializeField] private SurfaceAxis normalAxis = SurfaceAxis.Y;
        [SerializeField] private bool penetrationWhenNegative = true;

        private readonly object sampleLock = new object();
        private PoseSample latest;
        private bool hasSample;
        private float nextSampleTime;
        private long sampleIndex;

        public void SetRawPoint(Transform point) => rawPoint = point;
        private void Start() => nextSampleTime = Time.unscaledTime;

        private void Update()
        {
            if (sampleRateHz <= 0 || origin == null || visualPoint == null || rawPoint == null) return;
            float interval = 1f / sampleRateHz;
            float now = Time.unscaledTime;
            // 元実装と同じ上限10回の追いつき処理。Updateの実行周期を超える測定精度は保証しない。
            for (int guard = 0; now >= nextSampleTime && guard < 10; guard++)
            {
                Capture();
                nextSampleTime += interval;
            }
            if (now - nextSampleTime > 1f) nextSampleTime = now + interval;
        }

        private void Capture()
        {
            PoseSample sample = PoseMath.Measure(origin, visualPoint, rawPoint,
                normalAxis, penetrationWhenNegative, positionScale,
                sampleIndex++, Time.realtimeSinceStartupAsDouble, contact != null && contact.IsTouching);
            lock (sampleLock)
            {
                latest = sample;
                hasSample = true;
            }
        }

        public bool TryGetLatestSample(out PoseSample sample)
        {
            lock (sampleLock) { sample = latest; return hasSample; }
        }
    }
}

namespace Portfolio.Research
{
    /// <summary>SpongeVisualizerから入力・フィルタ・描画の責務を分離したサンプル。</summary>
    public sealed class SpongeVisualizer : MonoBehaviour
    {
        [SerializeField] private ExperimentTcpBridge tcp;
        [SerializeField] private Renderer targetRenderer;
        [SerializeField] private float movementScale = 0.001f;
        [SerializeField] private Vector3 moveDirection = Vector3.forward;
        [SerializeField, Range(0f, 0.5f)] private float smoothTime = 0.1f;
        [SerializeField] private float noiseThreshold = 0.05f;
        [SerializeField] private Color normalColor = Color.white, nearZeroColor = Color.blue, activeColor = Color.red;
        [SerializeField] private bool useNearZeroColor = true;
        [SerializeField] private float nearZeroThreshold = 1f, activeThreshold = 80f;
        [Header("実機なしの表示確認用。研究計測値ではありません")]
        [SerializeField] private bool useDemoInput;
        [SerializeField] private float demoDisplacementMm;

        private readonly DisplacementSignal signal = new DisplacementSignal();
        private Vector3 initialPosition, velocity;

        private void Start()
        {
            initialPosition = transform.localPosition;
            if (targetRenderer == null) targetRenderer = GetComponent<Renderer>();
            if (!useDemoInput && tcp == null) Debug.LogWarning("TCP参照を設定してください。", this);
            UpdateColor(useDemoInput ? demoDisplacementMm : tcp != null ? tcp.Displacement : 0f);
        }

        private void Update()
        {
            if (!useDemoInput && tcp == null) return;
            float raw = useDemoInput ? demoDisplacementMm : tcp.Displacement;
            float filtered = signal.Observe(raw, noiseThreshold);
            Vector3 target = initialPosition + moveDirection * filtered * movementScale;
            transform.localPosition = Vector3.SmoothDamp(transform.localPosition, target, ref velocity, smoothTime);
            UpdateColor(raw);
        }

        private void UpdateColor(float raw)
        {
            if (targetRenderer == null) return;
            DisplacementBand band = DisplacementSignal.Classify(raw, useNearZeroColor, nearZeroThreshold, activeThreshold);
            Color color = band == DisplacementBand.NearZero ? nearZeroColor :
                          band == DisplacementBand.Active ? activeColor : normalColor;
            if (targetRenderer.material.color != color) targetRenderer.material.color = color;
        }
    }
}

using System;
using System.Collections.Generic;
using UnityEngine;

namespace Portfolio.Research
{
    public enum ExperimentRunState
    {
        NotStarted,
        Presenting,
        WaitingForNextSet,
        Finished
    }

    public sealed class ExperimentSequence : MonoBehaviour
    {
        [Header("References")]
        [SerializeField] private ExperimentTcpBridge tcpBridge;
        [SerializeField] private SetCsvStore csvStore;

        [Header("Experiment Settings")]
        [Tooltip("実験開始時に全セットの条件順・CSVを準備します。開始後の設定変更は次の実験から反映されます。")]
        [SerializeField, Min(1)] private int totalSets = 5;
        [Tooltip("剛性条件の番号。初期値は2,3,4。正の整数で重複不可。")]
        [SerializeField] private List<int> stiffnessIds = new List<int> { 2, 3, 4 };
        [Tooltip("振動条件の番号。初期値は1,2。全剛性との組み合わせを1セットにします。0も使用可能。")]
        [SerializeField] private List<int> vibrationIds = new List<int> { 1, 2 };
        [Tooltip("0なら実験開始時に自動生成します。0以外なら同じ順番を再現できます。")]
        [SerializeField] private int randomSeed;

        [Header("Runtime Status (Read Only)")]
        [SerializeField, HideInInspector] private ExperimentRunState state = ExperimentRunState.NotStarted;
        [SerializeField, HideInInspector] private int sessionSeed;
        [SerializeField, HideInInspector] private int currentSet;
        [SerializeField, HideInInspector] private int currentTrial;
        [SerializeField, HideInInspector] private int currentConditionId;
        [SerializeField, HideInInspector] private List<int> currentOrder = new List<int>();

        public event Action CurrentTrialChanged;
        private readonly List<List<int>> plannedOrders = new List<List<int>>();
        private List<ExperimentCondition> activeConditions = new List<ExperimentCondition>();

        public ExperimentRunState State => state;
        public int TotalSets => plannedOrders.Count > 0 ? plannedOrders.Count : totalSets;
        public int SessionSeed => sessionSeed;
        public int CurrentSet => currentSet;
        public int CurrentTrial => currentTrial;
        public int CurrentConditionId => currentConditionId;
        public int CurrentStiffnessId => GetConditionDefinition(currentConditionId).StiffnessId;
        public int CurrentVibrationId => GetConditionDefinition(currentConditionId).VibrationId;
        public IReadOnlyList<int> CurrentOrder => currentOrder;

        public ExperimentCondition GetConditionDefinition(int id)
        {
            return id >= 1 && id <= activeConditions.Count ? activeConditions[id - 1] : default;
        }

        private void OnEnable()
        {
            if (tcpBridge != null)
            {
                tcpBridge.NextRequested += HandleNextRequest;
            }
        }

        private void OnDisable()
        {
            if (tcpBridge != null)
            {
                tcpBridge.NextRequested -= HandleNextRequest;
            }
        }

        public void StartExperiment()
        {
            if (state != ExperimentRunState.NotStarted && state != ExperimentRunState.Finished)
            {
                Debug.LogWarning("[ExperimentSequence] Experiment is already running.", this);
                return;
            }

            if (!ValidateSettings())
            {
                return;
            }

            sessionSeed = randomSeed == 0 ? Environment.TickCount : randomSeed;
            activeConditions = ExperimentCondition.BuildCombinations(stiffnessIds, vibrationIds);
            plannedOrders.Clear();
            plannedOrders.AddRange(ConditionSchedule.Create(activeConditions, totalSets, sessionSeed));
            currentSet = 1;
            csvStore.BeginSession();
            // Prepare every answer target before presentation; later set starts
            // must reuse these rows instead of clearing answers entered in advance.
            for (int set = 1; set <= plannedOrders.Count; set++)
            {
                var conditions = new List<ExperimentCondition>();
                foreach (int id in plannedOrders[set - 1]) conditions.Add(GetConditionDefinition(id));
                csvStore.CreateSet(set, conditions);
            }
            BeginCurrentSet();
        }

        public void StartNextSet()
        {
            if (state != ExperimentRunState.WaitingForNextSet)
            {
                Debug.LogWarning("[ExperimentSequence] Next set can only start while waiting.", this);
                return;
            }

            if (currentSet >= TotalSets)
            {
                state = ExperimentRunState.Finished;
                return;
            }

            currentSet++;
            BeginCurrentSet();
        }

        public void NextTrial()
        {
            if (state != ExperimentRunState.Presenting)
            {
                return;
            }

            int currentIndex = currentTrial - 1;
            if (currentIndex + 1 < currentOrder.Count)
            {
                currentTrial++;
                ApplyCurrentCondition();
                return;
            }

            CompleteCurrentSet();
        }

        public void PreviousTrial()
        {
            if (state != ExperimentRunState.Presenting || currentTrial <= 1)
            {
                return;
            }

            currentTrial--;
            ApplyCurrentCondition();
        }

        public void JumpToTrial(int trialNumber)
        {
            if (state != ExperimentRunState.Presenting ||
                trialNumber < 1 || trialNumber > currentOrder.Count)
            {
                return;
            }

            currentTrial = trialNumber;
            ApplyCurrentCondition();
        }

        public void ReapplyCurrentCondition()
        {
            if (state == ExperimentRunState.Presenting)
            {
                ApplyCurrentCondition();
            }
        }

        public void ReopenLastTrial()
        {
            if ((state != ExperimentRunState.WaitingForNextSet &&
                 state != ExperimentRunState.Finished) ||
                currentOrder.Count == 0)
            {
                return;
            }

            state = ExperimentRunState.Presenting;
            currentTrial = currentOrder.Count;
            ApplyCurrentCondition();
        }

        private void BeginCurrentSet()
        {
            currentOrder = new List<int>(plannedOrders[currentSet - 1]);
            currentTrial = 1;
            state = ExperimentRunState.Presenting;

            ApplyCurrentCondition();

            Debug.Log(
                $"[ExperimentSequence] Set {currentSet} started. Order: {string.Join(", ", currentOrder)}",
                this);
        }

        private void ApplyCurrentCondition()
        {
            currentConditionId = currentOrder[currentTrial - 1];
            tcpBridge?.PublishCondition(CurrentStiffnessId, CurrentVibrationId);
            CurrentTrialChanged?.Invoke();

            Debug.Log(
                $"[ExperimentSequence] Set {currentSet}, Trial {currentTrial}, Condition {currentConditionId}, Stiffness {CurrentStiffnessId}, Vibration {CurrentVibrationId}",
                this);
        }

        private void CompleteCurrentSet()
        {
            currentConditionId = 0;
            tcpBridge?.PublishCondition(0, 0);
            CurrentTrialChanged?.Invoke();

            if (currentSet >= TotalSets)
            {
                state = ExperimentRunState.Finished;
                Debug.Log("[ExperimentSequence] Experiment finished.", this);
            }
            else
            {
                state = ExperimentRunState.WaitingForNextSet;
                Debug.Log(
                    $"[ExperimentSequence] Set {currentSet} finished. Start the next set from the Inspector.",
                    this);
            }
        }

        private void HandleNextRequest()
        {
            NextTrial();
        }

        private bool ValidateSettings()
        {
            if (csvStore == null)
            {
                Debug.LogError("[ExperimentSequence] SetCsvStore is not assigned.", this);
                return false;
            }

            try
            {
                ExperimentCondition.BuildCombinations(stiffnessIds, vibrationIds);
            }
            catch (ArgumentException exception)
            {
                Debug.LogError("[ExperimentSequence] " + exception.Message, this);
                return false;
            }

            return true;
        }

        private void OnValidate()
        {
            totalSets = Mathf.Max(1, totalSets);
        }
    }
}

using System;
using UnityEngine;

namespace Portfolio.Research
{
    public enum SurfaceAxis { X, Y, Z }

    [Serializable]
    public struct PoseSample
    {
        public long Index;
        public double TimeSeconds;
        public Vector3 VisualPositionMm, RawPositionMm;
        public Vector3 VisualEulerDegrees, RawEulerDegrees;
        public float PenetrationMm;
        public bool Contact;
    }

    public static class PoseMath
    {
        public static PoseSample Measure(Transform origin, Transform visual, Transform raw,
            SurfaceAxis axis, bool negativeIsPenetration, float scale,
            long index, double time, bool contact)
        {
            Vector3 visualLocal = origin.InverseTransformPoint(visual.position);
            Vector3 rawLocal = origin.InverseTransformPoint(raw.position);
            Quaternion inverseOrigin = Quaternion.Inverse(origin.rotation);
            float normal = axis == SurfaceAxis.X ? rawLocal.x :
                           axis == SurfaceAxis.Y ? rawLocal.y : rawLocal.z;
            return new PoseSample
            {
                Index = index, TimeSeconds = time,
                VisualPositionMm = visualLocal * scale,
                RawPositionMm = rawLocal * scale,
                VisualEulerDegrees = Normalize((inverseOrigin * visual.rotation).eulerAngles),
                RawEulerDegrees = Normalize((inverseOrigin * raw.rotation).eulerAngles),
                PenetrationMm = Mathf.Max(0f, negativeIsPenetration ? -normal : normal) * scale,
                Contact = contact
            };
        }

        private static Vector3 Normalize(Vector3 angles) =>
            new Vector3(Signed(angles.x), Signed(angles.y), Signed(angles.z));
        private static float Signed(float value) => value > 180f ? value - 360f : value;
    }
}

using System;

namespace Portfolio.Research
{
    public enum DisplacementBand { Normal, NearZero, Active }

    /// <summary>Unityに依存しないデッドバンド。色判定には必ず生の値を使う。</summary>
    public sealed class DisplacementSignal
    {
        private bool initialized;
        public float Filtered { get; private set; }

        public float Observe(float raw, float noiseThreshold)
        {
            if (!initialized || Math.Abs(raw - Filtered) > noiseThreshold)
            {
                Filtered = raw;
                initialized = true;
            }
            return Filtered;
        }

        public static DisplacementBand Classify(float raw, bool nearZeroEnabled,
            float nearZeroThreshold, float activeThreshold)
        {
            float absolute = Math.Abs(raw);
            if (nearZeroEnabled && absolute <= Math.Max(0, nearZeroThreshold))
                return DisplacementBand.NearZero;
            return absolute >= activeThreshold ? DisplacementBand.Active : DisplacementBand.Normal;
        }
    }
}

using System;
using System.Collections.Generic;

namespace Portfolio.Research
{
    /// <summary>元ExperimentSequenceのseed式とFisher–Yates順序を維持。</summary>
    public static class ConditionSchedule
    {
        public static List<List<int>> Create(IReadOnlyList<ExperimentCondition> conditions, int totalSets, int seed)
        {
            if (conditions == null || conditions.Count == 0) throw new ArgumentException("条件が必要です。");
            if (totalSets < 1) throw new ArgumentOutOfRangeException(nameof(totalSets));
            var sets = new List<List<int>>();
            for (int set = 1; set <= totalSets; set++)
            {
                var order = new List<int>();
                foreach (ExperimentCondition condition in conditions) order.Add(condition.Id);
                Shuffle(order, new Random(unchecked(seed + set * 1009)));
                sets.Add(order);
            }
            return sets;
        }

        private static void Shuffle<T>(IList<T> list, Random random)
        {
            for (int index = list.Count - 1; index > 0; index--)
            {
                int swap = random.Next(index + 1);
                T value = list[index];
                list[index] = list[swap];
                list[swap] = value;
            }
        }
    }
}

using System;
using System.Collections.Generic;

namespace Portfolio.Research
{
    // Internal ID identifies a combination; only the two factor IDs go over TCP.
    public readonly struct ExperimentCondition
    {
        public int Id { get; }
        public int StiffnessId { get; }
        public int VibrationId { get; }

        public ExperimentCondition(int id, int stiffnessId, int vibrationId)
        {
            Id = id;
            StiffnessId = stiffnessId;
            VibrationId = vibrationId;
        }

        public static List<ExperimentCondition> BuildCombinations(
            IReadOnlyList<int> stiffnessIds, IReadOnlyList<int> vibrationIds)
        {
            Validate(stiffnessIds, 1, "Stiffness IDs");
            Validate(vibrationIds, 0, "Vibration IDs");
            var result = new List<ExperimentCondition>();
            foreach (int stiffness in stiffnessIds)
                foreach (int vibration in vibrationIds)
                    result.Add(new ExperimentCondition(result.Count + 1, stiffness, vibration));
            return result;
        }

        private static void Validate(IReadOnlyList<int> ids, int minimum, string label)
        {
            if (ids == null || ids.Count == 0) throw new ArgumentException(label + " cannot be empty.");
            var used = new HashSet<int>();
            foreach (int id in ids)
                if (id < minimum || !used.Add(id))
                    throw new ArgumentException(label + " must be unique and >= " + minimum + ".");
        }
    }
}

using System;
using System.Collections.Concurrent;
using System.Globalization;
using System.IO;
using System.Net;
using System.Net.Sockets;
using System.Text;
using System.Threading;
using UnityEngine;

namespace Portfolio.Research
{
    /// <summary>
    /// dSPACEとのTCP通信だけを担当するクラス。
    /// 受信: displacement,button\n
    /// 送信: stiffnessId,vibrationId\n（0,0は停止）
    /// </summary>
    public sealed class ExperimentTcpBridge : MonoBehaviour
    {
        [Header("TCP Server")]
        [Tooltip("0.0.0.0なら、このPCのすべてのネットワークインターフェースで待ち受けます。")]
        [SerializeField] private string bindAddress = "0.0.0.0";
        [SerializeField, Min(1)] private int port = 12345;
        [SerializeField, Min(0.005f)] private float sendIntervalSeconds = 0.02f;

        [Header("dSPACE Button")]
        [SerializeField] private float buttonThreshold = 0.5f;
        [SerializeField, Min(0f)] private float debounceSeconds = 0.15f;

        [Header("Runtime Status (Read Only)")]
        [SerializeField] private float displacement;
        [SerializeField] private float buttonLevel;
        [SerializeField] private bool isListening;
        [SerializeField] private bool isClientConnected;
        [SerializeField] private int sentStiffnessId;
        [SerializeField] private int sentVibrationId;

        private readonly ConcurrentQueue<string> receivedLines = new ConcurrentQueue<string>();
        private readonly ConcurrentQueue<string> logMessages = new ConcurrentQueue<string>();
        private readonly object outputLock = new object();
        private readonly object socketLock = new object();

        private Thread serverThread;
        private TcpListener listener;
        private TcpClient activeClient;
        private volatile bool stopRequested;
        private bool lastButtonPressed;
        private float lastButtonEventTime = float.NegativeInfinity;

        public event Action NextRequested;

        public float Displacement => displacement;
        public float ButtonLevel => buttonLevel;
        public bool IsListening => isListening;
        public bool IsClientConnected => isClientConnected;
        public int SentStiffnessId => sentStiffnessId;
        public int SentVibrationId => sentVibrationId;

        private void OnEnable()
        {
            StartServer();
        }

        private void Update()
        {
            while (receivedLines.TryDequeue(out string line))
            {
                ProcessReceivedLine(line);
            }

            while (logMessages.TryDequeue(out string message))
            {
                Debug.Log(message, this);
            }
        }

        public void PublishCondition(int stiffnessId, int vibrationId)
        {
            lock (outputLock)
            {
                sentStiffnessId = stiffnessId;
                sentVibrationId = vibrationId;
            }
        }

        public void StartServer()
        {
            if (serverThread != null && serverThread.IsAlive)
            {
                return;
            }

            stopRequested = false;
            serverThread = new Thread(ServerLoop)
            {
                IsBackground = true,
                Name = "SI2026 Experiment TCP Server"
            };
            serverThread.Start();
        }

        public void StopServer()
        {
            stopRequested = true;

            lock (socketLock)
            {
                try { activeClient?.Close(); } catch { }
                try { listener?.Stop(); } catch { }
            }

            if (serverThread != null && serverThread.IsAlive)
            {
                serverThread.Join(500);
            }

            serverThread = null;
            isListening = false;
            isClientConnected = false;
        }

        private void ServerLoop()
        {
            try
            {
                IPAddress address = ResolveBindAddress(bindAddress);
                TcpListener localListener = new TcpListener(address, port);

                lock (socketLock)
                {
                    listener = localListener;
                }

                localListener.Start();
                isListening = true;
                logMessages.Enqueue($"[ExperimentTcpBridge] Listening on {address}:{port}");

                while (!stopRequested)
                {
                    if (!localListener.Pending())
                    {
                        Thread.Sleep(10);
                        continue;
                    }

                    TcpClient client = localListener.AcceptTcpClient();
                    lock (socketLock)
                    {
                        activeClient = client;
                    }

                    RunClientSession(client);

                    lock (socketLock)
                    {
                        if (ReferenceEquals(activeClient, client))
                        {
                            activeClient = null;
                        }
                    }
                }
            }
            catch (SocketException exception)
            {
                if (!stopRequested)
                {
                    logMessages.Enqueue($"[ExperimentTcpBridge] Socket error: {exception.Message}");
                }
            }
            catch (Exception exception)
            {
                if (!stopRequested)
                {
                    logMessages.Enqueue($"[ExperimentTcpBridge] Error: {exception}");
                }
            }
            finally
            {
                isListening = false;
                isClientConnected = false;
                lock (socketLock)
                {
                    try { activeClient?.Close(); } catch { }
                    try { listener?.Stop(); } catch { }
                    activeClient = null;
                    listener = null;
                }
            }
        }

        private void RunClientSession(TcpClient client)
        {
            try
            {
                client.NoDelay = true;
                isClientConnected = true;
                logMessages.Enqueue($"[ExperimentTcpBridge] Connected: {client.Client.RemoteEndPoint}");

                using (client)
                using (NetworkStream stream = client.GetStream())
                {
                    byte[] receiveBuffer = new byte[2048];
                    StringBuilder lineBuffer = new StringBuilder();
                    int sleepMilliseconds = Math.Max(
                        1,
                        (int)Math.Round(
                            sendIntervalSeconds * 1000f,
                            MidpointRounding.AwayFromZero));

                    while (!stopRequested && client.Connected)
                    {
                        if (stream.DataAvailable)
                        {
                            int length = stream.Read(receiveBuffer, 0, receiveBuffer.Length);
                            if (length <= 0)
                            {
                                break;
                            }

                            AppendReceivedText(lineBuffer, Encoding.UTF8.GetString(receiveBuffer, 0, length));
                        }

                        int stiffnessId;
                        int vibrationId;
                        lock (outputLock)
                        {
                            stiffnessId = sentStiffnessId;
                            vibrationId = sentVibrationId;
                        }

                        string message = stiffnessId.ToString(CultureInfo.InvariantCulture) + "," +
                            vibrationId.ToString(CultureInfo.InvariantCulture) + "\n";
                        byte[] sendBytes = Encoding.UTF8.GetBytes(message);
                        stream.Write(sendBytes, 0, sendBytes.Length);

                        Thread.Sleep(sleepMilliseconds);
                    }
                }
            }
            catch (Exception exception) when (
                exception is IOException ||
                exception is SocketException ||
                exception is ObjectDisposedException)
            {
                if (!stopRequested)
                {
                    logMessages.Enqueue($"[ExperimentTcpBridge] Client disconnected: {exception.Message}");
                }
            }
            finally
            {
                isClientConnected = false;
            }
        }

        private void AppendReceivedText(StringBuilder lineBuffer, string text)
        {
            lineBuffer.Append(text);

            while (true)
            {
                int newlineIndex = FindNewline(lineBuffer);
                if (newlineIndex < 0)
                {
                    break;
                }

                string line = lineBuffer.ToString(0, newlineIndex).TrimEnd('\r');
                lineBuffer.Remove(0, newlineIndex + 1);
                if (!string.IsNullOrWhiteSpace(line))
                {
                    receivedLines.Enqueue(line);
                }
            }
        }

        private static int FindNewline(StringBuilder text)
        {
            for (int index = 0; index < text.Length; index++)
            {
                if (text[index] == '\n')
                {
                    return index;
                }
            }

            return -1;
        }

        private void ProcessReceivedLine(string line)
        {
            string[] values = line.Split(',');
            if (values.Length < 2 ||
                !TryParseFloat(values[0], out float receivedDisplacement) ||
                !TryParseFloat(values[1], out float receivedButton))
            {
                Debug.LogWarning($"[ExperimentTcpBridge] Invalid receive line: {line}", this);
                return;
            }

            displacement = receivedDisplacement;
            buttonLevel = receivedButton;

            bool isPressed = receivedButton >= buttonThreshold;
            if (isPressed && !lastButtonPressed &&
                Time.unscaledTime - lastButtonEventTime >= debounceSeconds)
            {
                lastButtonEventTime = Time.unscaledTime;
                NextRequested?.Invoke();
            }

            lastButtonPressed = isPressed;
        }

        private static bool TryParseFloat(string text, out float value)
        {
            return float.TryParse(
                text.Trim(),
                NumberStyles.Float,
                CultureInfo.InvariantCulture,
                out value);
        }

        private static IPAddress ResolveBindAddress(string address)
        {
            if (string.IsNullOrWhiteSpace(address) || address == "0.0.0.0" || address == "*")
            {
                return IPAddress.Any;
            }

            if (!IPAddress.TryParse(address, out IPAddress parsedAddress))
            {
                throw new FormatException($"Invalid bind address: {address}");
            }

            return parsedAddress;
        }

        private void OnDisable()
        {
            StopServer();
        }

        private void OnApplicationQuit()
        {
            StopServer();
        }

        private void OnValidate()
        {
            port = Mathf.Clamp(port, 1, 65535);
            sendIntervalSeconds = Mathf.Max(0.005f, sendIntervalSeconds);
            debounceSeconds = Mathf.Max(0f, debounceSeconds);
        }
    }

    // Only persisted scores belong here; the selection cursor's default 4 is not an answer.

        public static string FormatSet(int questionCount,
            IEnumerable<(int Set, int Trial, int[] Scores)> rows, int setNumber)
        {
            return Format(questionCount, rows.Where(row => row.Set == setNumber));
        }

        public static string Format(int questionCount,
            IEnumerable<(int Set, int Trial, int[] Scores)> rows)
        {
            var text = new StringBuilder();
            text.Append("Set".PadRight(5)).Append(" | ").Append("Seq".PadRight(5));
            for (int i = 0; i < questionCount; i++)
                text.Append(" | ").Append(("Q" + (i + 1).ToString(CultureInfo.InvariantCulture)).PadRight(5));
            text.AppendLine();
            foreach (var row in rows)
            {
                text.Append(row.Set.ToString(CultureInfo.InvariantCulture).PadRight(5))
                    .Append(" | ").Append(row.Trial.ToString(CultureInfo.InvariantCulture).PadRight(5));
                for (int i = 0; i < questionCount; i++)
                {
                    int score = row.Scores != null && i < row.Scores.Length ? row.Scores[i] : 0;
                    text.Append(" | ").Append((score == 0 ? string.Empty :
                        score.ToString(CultureInfo.InvariantCulture)).PadRight(5));
                }
                text.AppendLine();
            }
            return text.ToString();
        }
    }

                return false;

                    return false;
                }

            int[] previousScores = row.Scores;
            string previousAnsweredAt = row.AnsweredAt;
            int previousRevision = row.Revision;
            row.Scores = scores.ToArray();
            row.AnsweredAt = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss", CultureInfo.InvariantCulture);
            row.Revision++;

            try
            {
                WriteSetFile(setNumber);
                cachedAnswerHistory = null;
                cachedSetHistory.Remove(setNumber);
                return true;

        private void WriteSetFile(int setNumber)
        {

            {
                return;

            StringBuilder csv = new StringBuilder();
            csv.Append("Trial,Condition,Stiffness,Vibration,AnsweredAt,Revision");
            foreach (SurveyQuestionDefinition question in activeQuestions)
            {
                csv.Append(',').Append(EscapeCsv(question.id));
            }
            csv.AppendLine();

            foreach (TrialRow row in rows)

                csv.Append(row.Trial.ToString(CultureInfo.InvariantCulture)).Append(',')
                    .Append(row.Condition.ToString(CultureInfo.InvariantCulture)).Append(',')
                    .Append(row.Stiffness.ToString(CultureInfo.InvariantCulture)).Append(',')
                    .Append(row.Vibration.ToString(CultureInfo.InvariantCulture)).Append(',')
                    .Append(EscapeCsv(row.AnsweredAt)).Append(',')
                    .Append(row.Revision.ToString(CultureInfo.InvariantCulture));

                foreach (int score in row.Scores)
                {
                    csv.Append(',');
                    if (score > 0)
                    {
                        csv.Append(score.ToString(CultureInfo.InvariantCulture));
                    }
                }
                csv.AppendLine();
            }

            string path = GetSetFilePath(setNumber);
            string temporaryPath = path + ".tmp";
            File.WriteAllText(temporaryPath, csv.ToString(), new UTF8Encoding(true));

            if (!File.Exists(path))
            {
                File.Move(temporaryPath, path);
                return;
            }

            try
            {
                File.Replace(temporaryPath, path, null);

            catch (PlatformNotSupportedException)
            {
                File.Copy(temporaryPath, path, true);
                File.Delete(temporaryPath);
            }

        private List<SurveyQuestionDefinition> BuildQuestionSnapshot()
        {
            List<SurveyQuestionDefinition> snapshot = questions
                .Where(question => question != null)
                .Select(question => question.Clone())
                .ToList();

            if (snapshot.Count == 0)
            {
                snapshot.Add(new SurveyQuestionDefinition());
            }

            HashSet<string> usedIds = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            for (int index = 0; index < snapshot.Count; index++)
            {
                string id = SanitizeColumnId(snapshot[index].id, $"Question{index + 1}");
                string uniqueId = id;
                int suffix = 2;
                while (!usedIds.Add(uniqueId))
                {
                    uniqueId = $"{id}_{suffix++}";
                }
                snapshot[index].id = uniqueId;
            }

            return snapshot;
        }

        private static string CreateUniqueDirectory(string root, string baseName)
        {
            string candidate = Path.Combine(root, baseName);
            int suffix = 2;
            while (Directory.Exists(candidate))
            {
                candidate = Path.Combine(root, $"{baseName}_{suffix++}");
            }
            return candidate;
        }

        private static string SanitizeFileName(string value, string fallback)
        {
            string result = string.IsNullOrWhiteSpace(value) ? fallback : value.Trim();
            foreach (char invalidCharacter in Path.GetInvalidFileNameChars())
            {
                result = result.Replace(invalidCharacter, '_');
            }
            return string.IsNullOrWhiteSpace(result) ? fallback : result;
        }

        private static string SanitizeColumnId(string value, string fallback)
        {
            if (string.IsNullOrWhiteSpace(value))
            {
                return fallback;
            }

            StringBuilder result = new StringBuilder();
            foreach (char character in value.Trim())
            {
                if (char.IsLetterOrDigit(character) || character == '_')
                {
                    result.Append(character);
                }
                else
                {
                    result.Append('_');
                }
            }

            return result.Length == 0 ? fallback : result.ToString();
        }

        private static string EscapeCsv(string value)
        {
            value ??= string.Empty;
            if (!value.Contains(",") && !value.Contains("\"") &&
                !value.Contains("\n") && !value.Contains("\r"))
            {
                return value;
            }

            return $"\"{value.Replace("\"", "\"\"")}\"";
        }
